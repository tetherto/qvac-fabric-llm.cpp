cmake_minimum_required(VERSION 3.20)
project(ggml_hybrid_package_test LANGUAGES C CXX)

if (HYBRID_PRODUCER)
    option(HYBRID_CORE_SHARED "Test a shared core with static CPU" OFF)
    set(BUILD_SHARED_LIBS ${HYBRID_CORE_SHARED} CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_STATIC ON CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    add_subdirectory(../../ggml ggml)
    get_target_property(cpu_type ggml-cpu TYPE)
    if (NOT cpu_type STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "Hybrid CPU must be a static library")
    endif()
    add_library(hybrid-module MODULE ../test-backend-loader-stub.cpp)
    target_include_directories(hybrid-module PRIVATE ../../ggml/src)
    target_link_libraries(hybrid-module PRIVATE ggml-base)
    target_compile_definitions(hybrid-module PRIVATE STUB_NAME="HYBRID_TEST" STUB_SCORE=1)
    install(TARGETS hybrid-module LIBRARY DESTINATION test-modules RUNTIME DESTINATION test-modules)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-name-$<CONFIG>.txt" CONTENT "$<TARGET_FILE_NAME:hybrid-module>")
else()
    find_package(ggml CONFIG REQUIRED)
    get_target_property(cpu_type ggml::ggml-cpu TYPE)
    if (NOT cpu_type STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "Installed CPU must be a static library")
    endif()
    add_executable(hybrid-consumer main.cpp)
    target_link_libraries(hybrid-consumer PRIVATE ggml::ggml)
    enable_testing()
    add_test(NAME cpu-without-modules COMMAND hybrid-consumer)
    add_test(NAME cpu-with-module COMMAND hybrid-consumer "${HYBRID_MODULE}")
endif()
