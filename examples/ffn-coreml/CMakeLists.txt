cmake_minimum_required(VERSION 3.16)
project(ffn-coreml LANGUAGES C CXX OBJC OBJCXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
option(FFN_COREML_SERVER "Build llama-server-coreml (needs FFN_COREML_DECODER_EXPERIMENT)" OFF)
if (FFN_COREML_SERVER)
    set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(../.. llama)

option(FFN_COREML_DECODER_EXPERIMENT "Enable the single-context decoder experiment" OFF)
if (FFN_COREML_DECODER_EXPERIMENT)
    target_compile_definitions(llama PRIVATE LLAMA_FFN_COREML_EXPERIMENT)
    target_include_directories(llama PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_executable(llama-coreml-decoder-bench decoder-coreml.mm)
    target_compile_definitions(ggml-metal PRIVATE GGML_METAL_FFN_EXPERIMENT)
    target_compile_definitions(llama-coreml-decoder-bench PRIVATE GGML_METAL_FFN_EXPERIMENT)
    target_include_directories(llama-coreml-decoder-bench PRIVATE ../../ggml/src ../../src)
    target_link_libraries(llama-coreml-decoder-bench PRIVATE ggml-metal)
    target_compile_options(llama-coreml-decoder-bench PRIVATE -fobjc-arc)
    target_link_libraries(llama-coreml-decoder-bench PRIVATE llama ggml "-framework CoreML" "-framework Foundation")
    if (FFN_COREML_SERVER)
        add_executable(llama-server-coreml server-coreml.mm)
        target_compile_definitions(llama-server-coreml PRIVATE GGML_METAL_FFN_EXPERIMENT)
        target_include_directories(llama-server-coreml PRIVATE ../../ggml/src ../../src)
        target_compile_options(llama-server-coreml PRIVATE -fobjc-arc)
        target_link_libraries(llama-server-coreml PRIVATE llama-server-impl ggml-metal llama ggml "-framework CoreML" "-framework Foundation")
    endif()
endif()

add_executable(llama-ffn-coreml ffn-coreml.mm)
target_compile_options(llama-ffn-coreml PRIVATE -fobjc-arc)
target_link_libraries(llama-ffn-coreml PRIVATE ggml "-framework CoreML" "-framework Foundation")
