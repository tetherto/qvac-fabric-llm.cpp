if (NOT APPLE)
    message(FATAL_ERROR "Core ML embedding requires macOS 15 or iOS 18")
endif()
find_library(COREML_FRAMEWORK CoreML REQUIRED)
find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
add_library(llama-embedding-coreml STATIC embedding-coreml.mm)
target_include_directories(llama-embedding-coreml PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(llama-embedding-coreml PRIVATE ggml-base ${COREML_FRAMEWORK} ${FOUNDATION_FRAMEWORK})
target_compile_features(llama-embedding-coreml PRIVATE cxx_std_17)
target_compile_options(llama-embedding-coreml PRIVATE -fobjc-arc)
set_target_properties(llama-embedding-coreml PROPERTIES POSITION_INDEPENDENT_CODE ON OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
if (LLAMA_BUILD_EXAMPLES)
    find_package(Threads REQUIRED)
    add_executable(llama-coreml-embedding-bench embedding-coreml-bench.cpp)
    target_link_libraries(llama-coreml-embedding-bench PRIVATE llama-embedding-coreml llama Threads::Threads)
    target_compile_features(llama-coreml-embedding-bench PRIVATE cxx_std_17)
    add_executable(llama-coreml-embedding-plan embedding-coreml-plan.m)
    target_link_libraries(llama-coreml-embedding-plan PRIVATE ${COREML_FRAMEWORK} ${FOUNDATION_FRAMEWORK})
    target_compile_options(llama-coreml-embedding-plan PRIVATE -fobjc-arc)
endif()
